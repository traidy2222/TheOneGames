cmake_minimum_required(VERSION 3.24)
project(TheOneGames VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(THEONE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../TheOne" CACHE PATH "External TheOne engine checkout")
if(NOT EXISTS "${THEONE_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Set THEONE_ROOT to the separate TheOne engine checkout")
endif()
# Reuse already downloaded, pinned dependency sources, never the engine's build outputs.
foreach(DEP SDL3 Jolt imgui cgltf json stb glm assimp meshoptimizer)
  string(TOLOWER "${DEP}" LOWER)
  string(TOUPPER "${DEP}" UPPER)
  if(EXISTS "${THEONE_ROOT}/build/_deps/${LOWER}-src/CMakeLists.txt" OR EXISTS "${THEONE_ROOT}/build/_deps/${LOWER}-src/imgui.h" OR EXISTS "${THEONE_ROOT}/build/_deps/${LOWER}-src/cgltf.h" OR EXISTS "${THEONE_ROOT}/build/_deps/${LOWER}-src/stb_image.h")
    set(FETCHCONTENT_SOURCE_DIR_${UPPER} "${THEONE_ROOT}/build/_deps/${LOWER}-src" CACHE PATH "Cached dependency source")
  endif()
endforeach()
set(THEONE_BUILD_TESTS OFF CACHE BOOL "Engine tests are verified separately")
set(THEONE_BUILD_TOOLS OFF CACHE BOOL "Generic tools are built in the engine")
add_subdirectory("${THEONE_ROOT}" "engine" EXCLUDE_FROM_ALL)
enable_testing()
add_library(game_import_host STATIC shared/import-host/application.cpp shared/import-host/visible_profile.cpp shared/import-host/frame_boundary.cpp shared/import-host/background_loading.cpp shared/import-host/scene_handoff.cpp shared/import-host/sandbox_game.cpp)
target_include_directories(game_import_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/shared/import-host")
target_link_libraries(game_import_host PUBLIC TheOne::Content TheOne::Renderer)
set(IMPORT_RUNTIME_DIRECTORY "${CMAKE_BINARY_DIR}/bin" CACHE PATH "Import-game executable root")
foreach(GAME sandbox robots)
  add_executable(${GAME} games/${GAME}/src/main.cpp)
  target_link_libraries(${GAME} PRIVATE game_import_host)
  set_target_properties(${GAME} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${IMPORT_RUNTIME_DIRECTORY}/${GAME}")
  theone_copy_shaders(${GAME})
  add_custom_command(TARGET ${GAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/games/sandbox/assets" "$<TARGET_FILE_DIR:${GAME}>/assets")
endforeach()
add_executable(game_integration_tests tests/game_integration.cpp)
target_link_libraries(game_integration_tests PRIVATE game_import_host)
add_test(NAME game_integration COMMAND game_integration_tests "${CMAKE_CURRENT_SOURCE_DIR}/games/sandbox/assets")
add_executable(robots_check_rig games/robots/tools/check_robots_rig.cpp)
target_link_libraries(robots_check_rig PRIVATE TheOne::Content)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME robots_animation_inventory COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/games/robots/tests/test_robots_animation_inventory.py")
add_test(NAME robots_motion_sampler COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/games/robots/tests/test_robots_motion.py")
add_test(NAME game_engine_separation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_separation.py" "${THEONE_ROOT}")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/verification/sandbox" "${CMAKE_BINARY_DIR}/verification/robots")
add_test(NAME sandbox_vulkan COMMAND sandbox --smoke-test)
set_tests_properties(sandbox_vulkan PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/verification/sandbox")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/games/robots/build/robots-village.one")
  add_test(NAME robots_vulkan COMMAND robots --smoke-test --scene "${CMAKE_CURRENT_SOURCE_DIR}/games/robots/build/robots-village.one")
  set_tests_properties(robots_vulkan PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/verification/robots")
endif()

add_library(planet_world STATIC games/planets/src/planet.cpp games/planets/src/mesh.cpp)
target_include_directories(planet_world PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/games/planets/src")
target_link_libraries(planet_world PUBLIC TheOne::Core nlohmann_json::nlohmann_json)
add_executable(planets games/planets/src/main.cpp)
target_link_libraries(planets PRIVATE planet_world TheOne::Renderer)
set(PLANET_RUNTIME_DIRECTORY "${CMAKE_BINARY_DIR}/bin/planets" CACHE PATH "Planet executable directory")
set_target_properties(planets PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PLANET_RUNTIME_DIRECTORY}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/launch/planets-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:planets>")
theone_copy_shaders(planets)
set(PLANET_FRAGMENT "${CMAKE_BINARY_DIR}/planet-shaders/scene.frag.spv")
add_custom_command(OUTPUT "${PLANET_FRAGMENT}"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/planet-shaders"
  COMMAND "${Vulkan_GLSLC_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/games/planets/shaders/scene.frag" "-I${THEONE_ROOT}/shaders" -o "${PLANET_FRAGMENT}"
  DEPENDS games/planets/shaders/scene.frag "${THEONE_ROOT}/shaders/scene_uniform.glsl" VERBATIM)
add_custom_target(planet_shader
  COMMAND ${CMAKE_COMMAND} -E make_directory "${PLANET_RUNTIME_DIRECTORY}/$<CONFIG>/shaders"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PLANET_FRAGMENT}" "${PLANET_RUNTIME_DIRECTORY}/$<CONFIG>/shaders/scene.frag.spv"
  DEPENDS "${PLANET_FRAGMENT}" VERBATIM)
add_dependencies(planet_shader planets_shaders)
add_dependencies(planets planet_shader)
# Override only this game's fragment shader after the common engine shader copy.
add_custom_command(TARGET planets POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${PLANET_FRAGMENT}" "$<TARGET_FILE_DIR:planets>/shaders/scene.frag.spv" VERBATIM)
set_property(TARGET planets APPEND PROPERTY LINK_DEPENDS "${PLANET_FRAGMENT}")
add_executable(planet_tests games/planets/tests/planet_tests.cpp)
target_link_libraries(planet_tests PRIVATE planet_world)
add_test(NAME planet_world COMMAND planet_tests)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/verification/planets-distance")
add_test(NAME planet_distance COMMAND planets --distance-smoke-test)
set_tests_properties(planet_distance PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/verification/planets-distance")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/verification/planets-async")
add_test(NAME planet_async COMMAND planets --async-smoke-test)
set_tests_properties(planet_async PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/verification/planets-async")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/verification/planets")
add_test(NAME planet_vulkan COMMAND planets --smoke-test)
set_tests_properties(planet_vulkan PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/verification/planets")
get_filename_component(THEONE_VULKAN_SDK "${Vulkan_INCLUDE_DIR}" DIRECTORY)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/games/sandbox/build/external/compatibility.one")
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/verification/compatibility")
  add_test(NAME sample_compatibility COMMAND sandbox --smoke-test
    --scene "${CMAKE_CURRENT_SOURCE_DIR}/games/sandbox/build/external/compatibility.one")
  set_tests_properties(sample_compatibility PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE
    WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/verification/compatibility")
endif()
# FindVulkan executes in the engine subdirectory, so consult its cached include path.
if(EXISTS "${THEONE_VULKAN_SDK}/Bin/VkLayer_khronos_validation.json")
  foreach(GAME sandbox robots planets)
    if(GAME STREQUAL "robots")
      if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/games/robots/build/robots-village.one")
        continue()
      endif()
      set(ARGS --scene "${CMAKE_CURRENT_SOURCE_DIR}/games/robots/build/robots-village.one")
    else()
      set(ARGS)
    endif()
    add_test(NAME ${GAME}_validation COMMAND ${GAME} --smoke-test --validate ${ARGS})
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/verification/${GAME}-validation")
    set_tests_properties(${GAME}_validation PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE
      WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/verification/${GAME}-validation"
      ENVIRONMENT "VK_LAYER_PATH=${THEONE_VULKAN_SDK}/Bin")
  endforeach()
endif()

add_executable(scene_handoff_profile tests/scene_handoff.cpp)
target_link_libraries(scene_handoff_profile PRIVATE game_import_host)
theone_copy_shaders(scene_handoff_profile)
add_test(NAME scene_handoff_focused COMMAND scene_handoff_profile "${CMAKE_CURRENT_SOURCE_DIR}/games/sandbox/assets/sandbox.one")
set_tests_properties(scene_handoff_focused PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/verification/sandbox" ENVIRONMENT "VK_LAYER_PATH=${THEONE_VULKAN_SDK}/Bin;VK_LAYER_ENABLES=VK_VALIDATION_FEATURE_ENABLE_SYNCHRONIZATION_VALIDATION_EXT")
add_executable(runtime_profile tests/runtime_profile.cpp)
target_link_libraries(runtime_profile PRIVATE game_import_host)
theone_copy_shaders(runtime_profile)


add_subdirectory(games/transformers)
add_subdirectory(games/tornado)

add_subdirectory(games/effects-lab)
